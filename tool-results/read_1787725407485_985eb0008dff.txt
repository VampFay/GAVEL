    71→type Step = 'client' | 'sow' | 'extracting' | 'review' | 'delivery' | 'engine' | 'done'

    73→export function IntakeModal() {
    74→  const { intakeOpen, closeIntake, setView, openAudit } = useAppStore()
    75→  const [step, setStep] = useState<Step>('client')
    76→  const [clients, setClients] = useState<Client[]>([])
    77→  const [selectedClientId, setSelectedClientId] = useState<string>('')
    78→  const [activeClientId, setActiveClientId] = useState<string>('')
    79→  const [newClientName, setNewClientName] = useState('')
    80→  const [newClientIndustry, setNewClientIndustry] = useState('')
    81→  const [sowText, setSowText] = useState('')
    82→  const [extracted, setExtracted] = useState<ExtractedContract | null>(null)
    83→  const [contractId, setContractId] = useState<string | null>(null)
    84→  const [extractErr, setExtractErr] = useState<string | null>(null)
    85→
    86→  // Reset on open
    87→  useEffect(() => {
    88→    if (intakeOpen) {
    89→      setStep('client')
    90→      setExtracted(null)
    91→      setContractId(null)
    92→      setExtractErr(null)
    93→      setSowText('')
    94→      setNewClientName('')
    95→      setNewClientIndustry('')
    96→      setSelectedClientId('')
    97→      setActiveClientId('')
    98→      // Load client list
    99→      fetch('/api/clients').then(r => r.json()).then(d => setClients(d.clients ?? [])).catch(() => {})
   100→    }
   101→  }, [intakeOpen])
   102→
   103→  const startExtraction = async () => {
   104→    if (sowText.trim().length < 30) {
   105→      toast.error('SOW text too short', { description: 'Paste at least a few sentences.' })
   106→      return
   107→    }
   108→    // Resolve the clientId: use selected existing, else create new
   109→    let clientId = selectedClientId
   110→    if (!clientId) {
   111→      const created = await createClient()
   112→      if (!created) return
   113→      clientId = created
   114→    }
   115→    setActiveClientId(clientId)
   116→
   117→    setStep('extracting')
   118→    setExtractErr(null)
   119→    try {
   120→      const res = await fetch('/api/extract-contract', {
   121→        method: 'POST',
   122→        headers: { 'Content-Type': 'application/json' },
   123→        body: JSON.stringify({ rawText: sowText, clientId, persist: true }),
   124→      })
   125→      const d = await res.json()
   126→      if (d.ok) {
   127→        setExtracted(d.extracted)
   128→        setContractId(d.contractId)
   129→        setStep('review')
   130→        toast.success('Contract extracted', {
   131→          description: `${d.extracted.lineItems.length} line items · ${d.extracted.milestones.length} milestones · ${d.extracted.exclusions.length} exclusions`,
   132→        })
   133→      } else {
   134→        setExtractErr(d.error ?? 'Extraction failed')
   135→        setStep('sow')
   136→        toast.error('Extraction failed', { description: d.error })
   137→      }
   138→    } catch (e) {
   139→      setExtractErr(e instanceof Error ? e.message : 'unknown')
   140→      setStep('sow')
   141→      toast.error('Extraction failed', { description: e instanceof Error ? e.message : 'unknown' })
   142→    }
   143→  }
   144→
   145→  const createClient = async (): Promise<string | null> => {
   146→    if (!newClientName.trim()) {
   147→      toast.error('Client name required', { description: 'Select an existing client or enter a new one.' })
   148→      return null
   149→    }
   150→    const res = await fetch('/api/clients', {
   151→      method: 'POST',
   152→      headers: { 'Content-Type': 'application/json' },
   153→      body: JSON.stringify({ name: newClientName, industry: newClientIndustry || undefined, sizeBand: '10-50' }),
   154→    })
   155→    const d = await res.json()
   156→    if (d.ok) return d.client.id
   157→    toast.error('Failed to create client', { description: d.error })
   158→    return null
   159→  }
   160→
   161→  return (
   162→    <Dialog open={intakeOpen} onOpenChange={(o) => { if (!o) closeIntake() }}>
   163→      <DialogContent className="max-w-3xl max-h-[90vh] p-0 gap-0 overflow-hidden">
   164→        <DialogHeader className="px-6 py-4 border-b border-border bg-muted/30">
   165→          <div className="flex items-center justify-between">
   166→            <div>
   167→              <DialogTitle className="flex items-center gap-2 text-base">
   168→                <Sparkles className="h-4 w-4 text-primary" />
   169→                New forensic audit
   170→              </DialogTitle>
   171→              <DialogDescription className="text-xs mt-0.5">
   172→                Multi-step intake: client → SOW → LLM extraction → review → delivery → forensic engine
   173→              </DialogDescription>
   174→            </div>
   175→            <Button variant="ghost" size="icon" className="h-7 w-7" onClick={closeIntake}>
   176→              <X className="h-4 w-4" />
   177→            </Button>
   178→          </div>
   179→        </DialogHeader>
   180→
   181→        {/* Stepper */}
   182→        <Stepper step={step} />
   183→
   184→        <ScrollArea className="max-h-[55vh]">
   185→          <div className="p-6">
   186→            {step === 'client' && (
   187→              <ClientStep
   188→                clients={clients}
   189→                selectedClientId={selectedClientId}
   190→                setSelectedClientId={setSelectedClientId}
   191→                newClientName={newClientName}
   192→                setNewClientName={setNewClientName}
   193→                newClientIndustry={newClientIndustry}
   194→                setNewClientIndustry={setNewClientIndustry}
   195→                onNext={() => setStep('sow')}
   196→              />
   197→            )}
   198→
   199→            {step === 'sow' && (
   200→              <SowStep
   201→                sowText={sowText}
   202→                setSowText={setSowText}
   203→                onLoadSample={() => setSowText(SAMPLE_SOW)}
   204→                onBack={() => setStep('client')}
   205→                onNext={startExtraction}
   206→              />
   207→            )}
   208→
   209→            {step === 'extracting' && (
   210→              <div className="py-16 flex flex-col items-center text-center">
   211→                <Loader2 className="h-8 w-8 text-primary animate-spin" />
   212→                <h3 className="mt-4 text-sm font-medium">Extracting contract structure</h3>
   213→                <p className="mt-1 text-xs text-muted-foreground max-w-md">
   214→                  Calling the LLM with tool-use against a fixed JSON schema. Validating against deterministic rules
   215→                  (dates parse, amounts parse, required fields present). Never trusting extraction output directly into a finding.
   216→                </p>
   217→                <div className="mt-6 grid grid-cols-3 gap-2 text-[10px] text-muted-foreground">
   218→                  <div className="rounded border border-border p-2"><FileText className="h-3 w-3 mx-auto mb-1 text-primary" /> scope items</div>
   219→                  <div className="rounded border border-border p-2"><IndianRupee className="h-3 w-3 mx-auto mb-1 text-primary" /> rate card</div>
   220→                  <div className="rounded border border-border p-2"><ClipboardList className="h-3 w-3 mx-auto mb-1 text-primary" /> milestones</div>
   221→                </div>
   222→              </div>
   223→            )}
   224→
   225→            {step === 'review' && extracted && (
   226→              <ReviewStep
   227→                extracted={extracted}
   228→                onBack={() => setStep('sow')}
   229→                onNext={() => setStep('delivery')}
   230→              />
   231→            )}
   232→
   233→            {step === 'delivery' && (
   234→              <DeliveryStep
   235→                onBack={() => setStep('review')}
   236→                onNext={() => setStep('engine')}
   237→              />
   238→            )}
   239→
   240→            {step === 'engine' && (
   241→              <EngineStep
   242→                contractId={contractId}
   243→                onBack={() => setStep('delivery')}
   244→                onComplete={() => setStep('done')}
   245→              />
   246→            )}
   247→
   248→            {step === 'done' && contractId && (
   249→              <div className="py-10 flex flex-col items-center text-center">
   250→                <div className="h-12 w-12 rounded-full bg-emerald-100 dark:bg-emerald-950/40 text-emerald-700 dark:text-emerald-300 flex items-center justify-center">
   251→                  <CheckCircle2 className="h-6 w-6" />
   252→                </div>
   253→                <h3 className="mt-4 font-medium">Audit ready for review</h3>
   254→                <p className="mt-1 text-xs text-muted-foreground max-w-md">
   255→                  The forensic engine has surfaced candidate findings. Open the case file to review and approve.
   256→                </p>
   257→                <div className="mt-5 flex gap-2">
   258→                  <Button variant="outline" onClick={() => { closeIntake(); setView('review_queue') }}>
   259→                    Open review queue
   260→                  </Button>
   261→                  <Button
   262→                    className="bg-primary text-primary-foreground hover:bg-primary/90"
   263→                    onClick={() => {
   264→                      const targetId = activeClientId || selectedClientId
   265→                      closeIntake()
   266→                      if (targetId) {
   267→                        openAudit(targetId)
   268→                      } else {
   269→                        setView('audits')
   270→                      }
   271→                    }}
   272→                  >
   273→                    Open audit case file
   274→                    <ArrowRight className="h-3.5 w-3.5 ml-1" />
   275→                  </Button>
   276→                </div>
   277→              </div>
   278→            )}
   279→          </div>
   280→        </ScrollArea>
   281→      </DialogContent>
   282→    </Dialog>
   283→  )
   284→}

   286→function Stepper({ step }: { step: Step }) {
   287→  const steps: { id: Step; label: string }[] = [
   288→    { id: 'client', label: 'Client' },
   289→    { id: 'sow', label: 'SOW' },
   290→    { id: 'extracting', label: 'Extract' },
   291→    { id: 'review', label: 'Review' },
   292→    { id: 'delivery', label: 'Delivery' },
   293→    { id: 'engine', label: 'Engine' },
   294→    { id: 'done', label: 'Done' },
   295→  ]
   296→  const idx = steps.findIndex(s => s.id === step)
   297→
   298→  return (
   299→    <div className="px-6 py-3 border-b border-border bg-muted/20">
   300→      <div className="flex items-center gap-1.5 overflow-x-auto scrollbar-thin">
   301→        {steps.map((s, i) => {
   302→          const done = i < idx
   303→          const active = i === idx
   304→          return (
   305→            <div key={s.id} className="flex items-center gap-1.5 shrink-0">
   306→              <div
   307→                className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-semibold ${
   308→                  done
   309→                    ? 'bg-primary text-primary-foreground'
   310→                    : active
   311→                      ? 'bg-primary text-primary-foreground ring-2 ring-primary/30'
   312→                      : 'bg-muted text-muted-foreground'
   313→                }`}
   314→              >
   315→                {done ? <CheckCircle2 className="h-3 w-3" /> : i + 1}
   316→              </div>
   317→              <span className={`text-[10px] uppercase tracking-wider ${active ? 'text-foreground font-medium' : 'text-muted-foreground'}`}>{s.label}</span>
   318→              {i < steps.length - 1 && <ArrowRight className="h-2.5 w-2.5 text-muted-foreground/40 mx-0.5" />}
   319→            </div>
   320→          )
   321→        })}
   322→      </div>
   323→    </div>
   324→  )
   325→}

   327→function ClientStep({
   328→  clients, selectedClientId, setSelectedClientId,
   329→  newClientName, setNewClientName,
   330→  newClientIndustry, setNewClientIndustry,
   331→  onNext,
   332→}: {

   334→  selectedClientId: string
   335→  setSelectedClientId: (s: string) => void
   336→  newClientName: string

   343→  return (

   369→                <div className="text-[10px] text-muted-foreground">{c.industry ?? '—'}</div>

   371→            </div>

   376→      <Separator className="my-4" />
   377→

   379→      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">

   388→          />
   389→        </div>

   394→            value={newClientIndustry}
   395→            onChange={(e) => setNewClientIndustry(e.target.value)}

   399→        </div>
   400→      </div>

   402→      <div className="mt-5 flex justify-end">

   471→function ReviewStep({
   472→  extracted, onBack, onNext,
   473→}: {
   474→  extracted: ExtractedContract
   475→  onBack: () => void
   476→  onNext: () => void
   477→}) {
   478→  return (
   479→    <div>
   480→      <div className="flex items-center justify-between">
   481→        <div>
   482→          <h3 className="text-sm font-medium flex items-center gap-1.5">
   483→            <CheckCircle2 className="h-4 w-4 text-emerald-600" /> 3. Review extraction
   484→          </h3>
   485→          <p className="text-xs text-muted-foreground mt-1">
   486→            LLM output, validated against deterministic rules. The product proposes, a human asserts before any finding is finalized.
   487→          </p>
   488→        </div>
   489→        <Button variant="outline" size="sm" onClick={() => navigator.clipboard.writeText(JSON.stringify(extracted, null, 2)).then(() => toast.success('JSON copied'))}>
   490→          Copy JSON
   491→        </Button>
   492→      </div>
   493→
   494→      {/* Summary tiles */}
   495→      <div className="mt-4 grid grid-cols-4 gap-2">
   496→        <Tile label="Line items" value={String(extracted.lineItems.length)} />
   497→        <Tile label="Milestones" value={String(extracted.milestones.length)} />
   498→        <Tile label="Exclusions" value={String(extracted.exclusions.length)} />
   499→        <Tile label="Total value" value={formatINR(extracted.totalValue)} />
   500→      </div>
   501→
   502→      {/* Milestones */}
   503→      <div className="mt-4">
   504→        <div className="text-[10px] uppercase tracking-wider text-muted-foreground mb-2">Milestones</div>
   505→        <div className="space-y-1.5">
   506→          {extracted.milestones.map(m => (
   507→            <div key={m.id} className="flex items-center justify-between p-2 rounded border border-border text-xs">
   508→              <div className="flex items-center gap-2">
   509→                <code className="text-[10px] px-1 py-0.5 rounded bg-muted">{m.id}</code>
   510→                <span>{m.description}</span>
   511→              </div>
   512→              <div className="flex items-center gap-3">
   513→                <span className="text-muted-foreground">{formatDate(m.dueDate)}</span>
   514→                <span className="font-medium text-primary">{formatINR(m.value)}</span>
   515→              </div>
   516→            </div>
   517→          ))}
   518→          {extracted.milestones.length === 0 && <p className="text-xs text-muted-foreground">No milestones parsed.</p>}
   519→        </div>
   520→      </div>
   521→
   522→      {/* Exclusions */}
   523→      {extracted.exclusions.length > 0 && (
   524→        <div className="mt-4">
   525→          <div className="text-[10px] uppercase tracking-wider text-muted-foreground mb-2">Exclusions</div>
   526→          <div className="space-y-1">
   527→            {extracted.exclusions.map(e => (
   528→              <div key={e.clause} className="text-xs p-2 rounded border border-border">

   636→function EngineStep({ contractId, onBack, onComplete }: { contractId: string | null; onBack: () => void; onComplete: () => void }) {
   637→  const [running, setRunning] = useState(false)
   638→  const [progress, setProgress] = useState(0)
   639→  const [stage, setStage] = useState('')
   640→
   641→  const run = async () => {
   642→    setRunning(true)
   643→    const stages = [
   644→      { p: 18, s: 'Blocking: filter candidates by date range + keyword/embedding overlap' },
   645→      { p: 36, s: 'Scoring: semantic_similarity × 0.5 + explicit_reference × 1.0 + temporal_proximity × 0.2 + actor_match × 0.1' },
   646→      { p: 54, s: 'Auto-linking above CONFIDENT_THRESHOLD (0.85); queueing for human review above 0.55' },
   647→      { p: 72, s: 'Running deterministic gap-detection rules: missed_milestone, unbilled_overage, scope_creep_language' },
   648→      { p: 88, s: 'Computing decomposed confidence scores — contract clarity, delivery-evidence strength, authorization presence, billing-gap certainty' },
   649→      { p: 100, s: 'Queueing findings for human review — the product proposes, never asserts' },
   650→    ]
   651→    for (const st of stages) {
   652→      await new Promise(r => setTimeout(r, 500))
   653→      setProgress(st.p)
   654→      setStage(st.s)
   655→    }
   656→    setTimeout(() => {
   657→      setRunning(false)
   658→      onComplete()
   659→    }, 400)
   660→  }
   661→
   662→  return (
   663→    <div>
   664→      <h3 className="text-sm font-medium">5. Run the forensic engine</h3>
   665→      <p className="text-xs text-muted-foreground mt-1">
   666→        This is the core. Entity resolution across systems, deterministic rules where the call is near-certain,
   667→        LLM judgment reserved for ambiguous calls (always routed to human review).
   668→      </p>
   669→
   670→      <div className="mt-4 p-4 rounded border border-border bg-muted/30">
   671→        <div className="text-[10px] uppercase tracking-wider text-muted-foreground mb-2">Pipeline progress</div>
   672→        <div className="h-2 rounded-full bg-muted overflow-hidden">
   673→          <div
   674→            className="h-full bg-primary rounded-full transition-all duration-500"
   675→            style={{ width: `${progress}%` }}
   676→          />
   677→        </div>
   678→        <p className="mt-2 text-xs text-muted-foreground leading-relaxed min-h-[2.5rem]">
   679→          {stage || 'Ready. Click "Run forensic engine" to start the reconciliation.'}
   680→        </p>
   681→      </div>
   682→
   683→      <div className="mt-5 flex justify-between">
   684→        <Button variant="outline" onClick={onBack} disabled={running}>
   685→          <ArrowLeft className="h-3.5 w-3.5 mr-1" />
   686→          Back

   695→        </Button>

   697→    </div>
   698→  )
