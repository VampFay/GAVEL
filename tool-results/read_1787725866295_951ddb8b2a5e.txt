   286→   286→function Stepper({ step }: { step: Step }) {
   287→   287→  const steps: { id: Step; label: string }[] = [
   288→   288→    { id: 'client', label: 'Client' },
   289→   289→    { id: 'sow', label: 'SOW' },
   290→   290→    { id: 'extracting', label: 'Extract' },
   291→   291→    { id: 'review', label: 'Review' },
   292→   292→    { id: 'delivery', label: 'Delivery' },
   293→   293→    { id: 'engine', label: 'Engine' },
   294→   294→    { id: 'done', label: 'Done' },
   295→   295→  ]
   296→   296→  const idx = steps.findIndex(s => s.id === step)
   297→   297→
   298→   298→  return (
   299→   299→    <div className="px-6 py-3 border-b border-border bg-muted/20">
   300→   300→      <div className="flex items-center gap-1.5 overflow-x-auto scrollbar-thin">
   301→   301→        {steps.map((s, i) => {
   302→   302→          const done = i < idx
   303→   303→          const active = i === idx
   304→   304→          return (
   305→   305→            <div key={s.id} className="flex items-center gap-1.5 shrink-0">
   306→   306→              <div
   307→   307→                className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-semibold ${
   308→   308→                  done
   309→   309→                    ? 'bg-primary text-primary-foreground'
   310→   310→                    : active
   311→   311→                      ? 'bg-primary text-primary-foreground ring-2 ring-primary/30'
   312→   312→                      : 'bg-muted text-muted-foreground'
   313→   313→                }`}
   314→   314→              >
   315→   315→                {done ? <CheckCircle2 className="h-3 w-3" /> : i + 1}
   316→   316→              </div>
   317→   317→              <span className={`text-[10px] uppercase tracking-wider ${active ? 'text-foreground font-medium' : 'text-muted-foreground'}`}>{s.label}</span>
   318→   318→              {i < steps.length - 1 && <ArrowRight className="h-2.5 w-2.5 text-muted-foreground/40 mx-0.5" />}
   319→   319→            </div>
   320→   320→          )
   321→   321→        })}
   322→   322→      </div>
   323→   323→    </div>
   324→   324→  )
   325→   325→}

   640→   640→

   642→   642→    setRunning(true)
   643→   643→    const stages = [
   644→   644→      { p: 18, s: 'Blocking: filter candidates by date range + keyword/embedding overlap' },
   645→   645→      { p: 36, s: 'Scoring: semantic_similarity × 0.5 + explicit_reference × 1.0 + temporal_proximity × 0.2 + actor_match × 0.1' },
